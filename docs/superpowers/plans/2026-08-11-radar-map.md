- [ ] **Step 3: Write minimal implementation**

```typescript
import React from 'react';

interface RadarCardProps {
  lat: number;
  lon: number;
  displayName?: string;
}

export default function RadarCard({ lat, lon, displayName }: RadarCardProps) {
  const radarUrl = `https://www.rainviewer.com/map.html?loc=${lat},${lon},8&o=3&c=3&k=1&o=1&lm=1&layer=radar&sm=1&sn=1`;

  return (
    <div className="bg-white rounded-lg shadow p-4 mb-4">
      <h3 className="text-lg font-semibold mb-2">
        Radar {displayName ? `— ${displayName}` : ''}
      </h3>
      <div className="relative w-full h-80 overflow-hidden rounded">
        <iframe
          title={`Weather Radar for ${displayName || 'Location'}`}
          src={radarUrl}
          width="100%"
          height="100%"
          style={{ border: 0 }}
          loading="lazy"
        />
      </div>
    </div>
  );
}
```
